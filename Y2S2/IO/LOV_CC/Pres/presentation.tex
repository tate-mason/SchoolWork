\section{Motivation}
% ============================================================

\begin{frame}{Motivation: Data Privacy, Ad Targeting, and Consumer Agency}
  \begin{itemize}
    \item Concerns around data privacy and ad targeting
    \begin{itemize}
      \item Facebook's $5B$ FTC settlement
      \item Google ad tracking opt-out
    \end{itemize}
    \item Amazon anti-trust scrutiny
    \begin{itemize}
      \item Amazon accounts for almost $40\%$ of US e-commerce sales (Statista, 2023)
      \item Uses consumer data to target ads and build market share as both retailer and platform
    \end{itemize}
    \item Consumer agency over data (opt out)
    \begin{itemize}
      \item Apple iOS 14.5 update: $85\%$ opt-out rate upon rollout (Flurry Analytics, 2022)
      \item Facebook ad revenue down \$13bn in 2022 due to app tracking transparency (ATT) policy
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Addressing the Concern}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item How can we test the impact of policy interventions surrounding data tracking and ad targeting in a context like Amazon, where the firm has strong incentives to strategically target ads and promotions?
        \item Structural demand estimation is a powerful tool for policy analysis, but
              requires a model of consumer behavior and firm incentives
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{tcolorbox}[colback=red!5!white, colframe=UGARed, title=\textbf{Key Challenge}]
        How do we model consumer behavior in a way that captures the key drivers of consumption dynamics and firms' response?
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Why Does Variety Matter?}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Canonical logit models ignore consumer preferences for variety, which are
              empirically important 
        \item These preferences may shape consumer switching behavior, which in turn
              shapes firms' incentives to strategically target ads and promotions (i.e. to encourage switch to a higher markup product)
        \item Understanding the role of variety-seeking is valuable for modeling consumer behavior and evaluating the impact data privacy policies on both consumers and firms 
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{tcolorbox}[colback=red!5!white, colframe=UGARed, title=\textbf{Research Question}]
        How much do consumer tastes for novelty and habit influence consumption dynamics and firms' advertising strategies in a dynamic market?
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{What This Paper Does}
  \begin{enumerate}
    \item \hl{Model:} Derive a discrete-choice demand system where the
          utility of a product depends on both deviation from the consumer's
          choice history $\theta_{it}$ and product quality $X_{jt}$
    \vspace{4pt}
    \item \hl{Simulation:} Characterize consumer and firm behavior
          across preference regimes $(\beta_i, \gamma_i)$ and advertising
          specifications
    \vspace{4pt}
    \item \hl{Estimation (coming):} Bring model to Nielsen scanner data
          to recover structural parameters
    \vspace{4pt}
    \item \hl{Counterfactuals (coming):} Study advertising targeting policy and
          welfare implications
  \end{enumerate}
  \vspace{6pt}
  \textbf{Today's talk:} Model + simulation results
\end{frame}

\section{Model}
% ============================================================

\begin{frame}{Consumer's Problem}
  Consumer $i$ chooses product $j$ in period $t$ to maximize:
  \begin{equation*}
    U_{ijt} = \underbrace{\beta_{ij} X_{jt}}_{\text{characteristic component}}
              + \underbrace{\gamma_i \log(1 + \Sigma_{ijt}^2)}_{\text{novelty component}}
              + \underbrace{\kappa \, a_{jt}}_{\text{advertising}}
              - \underbrace{\alpha \, p_{jt}}_{\text{price}}
              + \xi_{j} + \varepsilon_{ijt}
  \end{equation*}
  \begin{itemize}
    \item where $\Sigma_{ijt} = X_{jt} - \theta_{it}$ is the deviation of product $j$'s characteristic from consumer $i$'s previously chosen characteristics $\theta_{it}$ in period $t$.
    \item at the moment, $\theta_{it} = \frac{1}{t}\sum_{s=1}^{t-1} X_{js}$, or the mean of previously chosen product characteristics.
  \end{itemize}
  \vspace{8pt}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \hl{Key parameters:}
      \begin{itemize}
        \item $\beta_i > 0$: marginal utility of quality
        \item $\gamma_i > 0$: variety-seeking
        \item $\kappa > 0$: advertising shifter
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \hl{Choice probability (MNL):}
      \begin{equation*}
        P_{ijt} = \frac{e^{U_{ijt}}}{1 + \sum_{k} e^{U_{ikt}}}
      \end{equation*}
      Follows from $\varepsilon_{ijt} \sim \text{Gumbel}(0,1)$
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{State Variable: History of Choices}
  \begin{tcolorbox}[colback=red!5!white, colframe=UGARed, title=\textbf{History of Choices $\theta_{it}$}]
    \[
    \theta_{it} = f(\{X_{js}\}_{s < t}) = \frac{1}{t}\sum_{s=1}^{t-1} X_{js}
    \]

    The consumer's history, for simplicity, is summarized by the mean of previously chosen product characteristics.
    Updates each period as the consumer makes a new choice.
    This is the key state variable linking past behavior to current utility.
  \end{tcolorbox}

  \vspace{6pt}
  \hl{Initialization strategie:}
  \begin{enumerate}
    \item \textbf{Initial state:} Simulate $T_{\text{prior}} = 5$ pre-sample
          periods from $J \in [1,2,3,4,5]$ to generate an experience-based
          prior. Similar to Bronnenberg, Dub\'e, Joo (2021).
  \end{enumerate}

  \vspace{1pt}
  \textit{Initial history yields more heterogeneous behavior --- more on this shortly.}
\end{frame}

\begin{frame}{Road to Estimation}
  \begin{enumerate}
    \item \hl{Multiple consumers} \\
          Extend simulation to $I = 5$ consumers who have heterogeneous $(\beta_i, \gamma_i)$;
          aggregate to market shares

    \vspace{4pt}
    \item \hl{Consumer value function} \\
          Move from reduced-form simulation to solving the full
          dynamic discrete choice with continuation values

    \vspace{4pt}
    \item \hl{Nielsen scanner data} \\
          Match simulated moments to data;
          identify $(\beta_i, \gamma_i)$ using BLP instruments

    \vspace{4pt}
    \item \hl{Firm's Problem} \\
          Ad targeting policy; profit maximization; counterfactuals
  \end{enumerate}
\end{frame}

\begin{frame}{Simulation Algorithm (Consumer)}
  \begin{enumerate}
    \item Draw $\varepsilon_{ijt} \sim \text{Gumbel}(0,1)$ for all $i, j, t$
    \item Initialize $\bar{X}_{i0}$ (set mean or prior mean)
    \item For $t = 1, \ldots, T$:
      \begin{enumerate}
        \item Compute $U_{ijt} = \beta_i X_{jt}
              - \gamma_i (X_{jt} - \bar{X}_{it})^2
              + \kappa a_{jt} + \varepsilon_{ijt}$
        \item Set $j^* = \arg\max_j U_{ijt}$
        \item Update $\bar{X}_{it+1} = \frac{1}{t}\sum_{k \le t} x_{ijk^*}$
      \end{enumerate}
    \item Compute CCPs via log-sum-exp for numerical stability:
          $\log P_{ijt} = U_{ijt} - \log \sum_k e^{U_{ikt}}$
    \item Average over $S = 1{,}000$ simulation draws
  \end{enumerate}
\end{frame}

\section{Motivation}  % ~3 min
%===============================================================================

\begin{frame}{Why Does Variety Matter?}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \textbf{Empirical facts:}
            \begin{itemize}
                \item Consumers switch brands even when quality is unchanged
                \item Yet brand preferences are highly persistent
                  \citep{bronnenberg2012brand}
                \item Advertising breaks inertia \citep{shum2004does}
            \end{itemize}
            \vspace{6pt}
            \textbf{These facts are hard to reconcile simultaneously} \\[4pt]
            $\Rightarrow$ Need a model that nests both variety-seeking
            and habit formation
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{block}{Research Question}
                How do consumer variety preferences shape the incentive
                to target advertising, and what does this imply for
                equilibrium market structure?
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{What This Paper Does}
    \begin{enumerate}
        \item \textbf{Model:} Derive a discrete-choice demand system where the
              disutility of a product depends on deviation from the consumer's
              experienced mean $\bar{X}_{it}$
        \vspace{4pt}
        \item \textbf{Simulation:} Characterize consumer and firm behavior
              across preference regimes $(\beta_i, \gamma_i)$ and advertising
              specifications
        \vspace{4pt}
        \item \textbf{Estimation (coming):} Bring model to Nielsen scanner data
              to recover structural parameters
        \vspace{4pt}
        \item \textbf{Counterfactuals (coming):} Study targeting policy and
              welfare implications
    \end{enumerate}
    \vspace{6pt}
    \textbf{Today's talk:} Model + simulation results
\end{frame}

\section{Model}  % ~5 min
%===============================================================================

\begin{frame}{Consumer's Problem}
    Consumer $i$ chooses product $j$ in period $t$ to maximize:
    \begin{equation*}
        U_{ijt} = \underbrace{\beta_i X_{jt}}_{\text{quality gain}}
                  - \underbrace{\gamma_i \Sigma_{ijt}^2}_{\text{variety penalty}}
                  + \underbrace{\kappa \, a_{jt}}_{\text{advertising}}
                  - \underbrace{\alpha \, p_{jt}}_{\text{price}}
                  + \xi_j + \varepsilon_{ijt}
    \end{equation*}
    where $\Sigma_{ijt} = X_{jt} - \bar{X}_{it}$ and
    $\bar{X}_{it} = \frac{1}{t}\sum_{k=1}^{t} x_{ikt}$

    \vspace{8pt}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Key parameters:}
            \begin{itemize}
                \item $\beta_i > 0$: marginal utility of quality
                \item $\gamma_i > 0$: variety-seeking
                \item $\gamma_i < 0$: habit persistence
                \item $\kappa > 0$: advertising shifter
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Choice probability (MNL):}
            \begin{equation*}
                P_{ijt} = \frac{e^{U_{ijt}}}{1 + \sum_{k} e^{U_{ikt}}}
            \end{equation*}
            Follows from $\varepsilon_{ijt} \sim \text{Gumbel}(0,1)$
            \citep{mcfadden1974conditional}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{State Variable: Experienced Mean}
    \begin{block}{Experienced Mean $\bar{X}_{it}$}
        \[
            \bar{X}_{it} = \frac{1}{t} \sum_{k=1}^{t} x_{ikt}
        \]
        Updates each period as the consumer makes a new choice.
        This is the key state variable linking past behavior to current utility.
    \end{block}

    \vspace{6pt}
    \textbf{Two initialization strategies (both simulated today):}
    \begin{enumerate}
        \item \textbf{Set mean:} $\bar{X}_{i0} = \bar{J}$, the mean of the
              product set. Simple benchmark.
        \item \textbf{Prior mean:} Simulate $T_{\text{prior}} = 5$ pre-sample
              periods from $J \sim |N(0,1)|$ to generate an experience-based
              prior. Inspired by \citet{bronnenberg2012brand}.
    \end{enumerate}

    \vspace{6pt}
    \textit{Prior mean yields more heterogeneous behavior — more on this shortly.}
\end{frame}

\begin{frame}{Firm's Problem}
    \textbf{Firm $j$ solves a dynamic program:}
    \[
        V_j(t; \Sigma_{jt}, \lambda_t(\gamma_i)) = \max_{a \in \{0,1\}}
        \left\{ \pi_{jt}(a) + \delta \, \E\bigl[V_j(t{+}1;\cdot)\bigr] \right\}
    \]

    \vspace{4pt}
    \textbf{Per-period profit:}
    \[
        \pi_{jt}(a) = s_{jt}(p - c) - \kappa \, a^2
    \]

    \vspace{4pt}
    \textbf{Two advertising specifications:}
    \begin{itemize}
        \item \textbf{Fixed:} $a \in \{0, 1\}$ (binary on/off)
        \item \textbf{Markovian:} $a_t \propto (\bar{X}_{it} - x_{jt-1})^2$
              — higher spending when product is far from consumer's mean
    \end{itemize}

    \vspace{4pt}
    Solved by backward induction over $T = 100$ periods.
\end{frame}

\begin{frame}{Set Mean: Flat Choice Probabilities}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \includegraphics[width=\linewidth]{figures/fixed_mean_choice_prob_product_3.pdf}
            % Replace with your actual figure path
        \end{column}
        \begin{column}{0.42\textwidth}
            \textbf{Key findings (set mean):}
            \begin{itemize}
                \item Probabilities are essentially flat: $\bar{X}_{it}$ is constant,
                      so the only period-to-period variation is $\varepsilon_{ijt}$
                \item $(\beta=1, \gamma=4)$: highest prob.\ for product 3 (closest
                      to mean)
                \item $(\beta=4, \gamma=1)$: gravitates to highest $X_j$
            \end{itemize}
            \vspace{4pt}
            \textit{Useful benchmark but behavioral dynamics absent.}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Prior Mean: Behavioral Dynamics Emerge}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \includegraphics[width=\linewidth]{figures/prior_mean_choice_prob_product_1.pdf}
            % Replace with your actual figure path
        \end{column}
        \begin{column}{0.42\textwidth}
            \textbf{Key findings (prior mean):}
            \begin{itemize}
                \item $(\beta=1, \gamma=2.5)$ flips across products ---
                      variety-seeking dominates
                \item $(\beta=4, \gamma=1)$ locks onto highest $X_j$ quickly
                \item $(\beta=4, \gamma=2.5)$ converges to product 4 after
                      $\sim$30 periods: penalty relevant early, dominated later
            \end{itemize}
            \vspace{4pt}
            \textbf{Takeaway:} A dynamic $\bar{X}_{it}$ is essential for
            realistic variety-seeking behavior.
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Regime Comparison: Probability Ordering}
    \begin{center}
        \small
        \renewcommand{\arraystretch}{1.3}
        \begin{tabular}{lcc}
            \toprule
            \textbf{Regime} & \textbf{Set Mean} & \textbf{Prior Mean} \\
            \midrule
            $(\beta=1, \gamma=4)$   & Concentrates at $X=3$ & Spreads across $X < 3$ \\
            $(\beta=4, \gamma=1)$   & Concentrates at $X=5$ & Locks onto highest product \\
            $(\beta=4, \gamma=2.5)$ & Moderate spread        & Converges after $\sim$30 $t$ \\
            $(\beta=1, \gamma=2.5)$ & Moderate spread        & Cycles among low-$X$ products \\
            \bottomrule
        \end{tabular}
    \end{center}

    \vspace{8pt}
    \begin{alertblock}{Key Insight}
        The prior mean specification generates the heterogeneous, dynamic
        sorting across products that we observe in scanner data, while the
        set mean is too static to be taken to data.
    \end{alertblock}
\end{frame}

    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \includegraphics[width=\linewidth]{figures/ad_choice_both.pdf}
            % Replace with your actual figure path
        \end{column}
        \begin{column}{0.42\textwidth}
            \textbf{Left panel:} Smoothed $\Pr(\text{advertise})$ over time \\[4pt]
            \textbf{Right panel:} $V(\text{ad}) - V(\text{no ad})$ over time

            \vspace{8pt}
            \begin{itemize}
                \item Ad incentive is high early (market building) and
                      declines as consumer's mean converges
                \item Markovian spec: incentive spikes whenever consumer
                      drifts far from firm's product
                \item High $\beta$ regimes: firm benefits more from advertising
                      (consumers are quality-sensitive)
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Markovian vs.\ Fixed Advertising}
    \begin{block}{Fixed Advertising ($a \in \{0,1\}$)}
        Firm chooses a constant binary action each period.
        Ad incentive $\approx$ constant once consumer probs stabilize.
    \end{block}

    \vspace{6pt}
    \begin{block}{Markovian Advertising ($a \propto (\bar{X}_{it} - x_{jt-1})^2$)}
        Firm advertises more intensively when the consumer's last choice
        is far from their product. More responsive to consumer state.
    \end{block}

    \vspace{6pt}
    \textbf{Implication for estimation:} The Markovian spec generates
    time-varying ad intensity that can be matched to observed promotional
    data in Nielsen, while the fixed spec cannot.
\end{frame}

\section{Next Steps}  % ~3 min
%===============================================================================

\begin{frame}{Road to Estimation}
    \begin{enumerate}
        \item \textbf{Multiple consumers} \\
              Extend simulation to $I = 5$ heterogeneous consumers;
              aggregate to market shares

        \vspace{4pt}
        \item \textbf{Heterogeneous $\gamma_i$} \\
              Draw $\gamma_i \sim N(0, \sigma_\gamma)$; simulate over
              $\sigma_\gamma \in \{0.5, 1.0, 1.5\}$

        \vspace{4pt}
        \item \textbf{Consumer value function} \\
              Move from reduced-form simulation to solving the full
              dynamic discrete choice with continuation values

        \vspace{4pt}
        \item \textbf{Nielsen scanner data} \\
              Match simulated moments to data;
              identify $(\beta_i, \gamma_i)$ using BLP instruments
              \citep{berry1995automobile, conlon2020best}

        \vspace{4pt}
        \item \textbf{Counterfactuals} \\
              Ad targeting policy; consumer welfare under variety vs.\ habit
    \end{enumerate}
\end{frame}

\begin{frame}{Simulation Algorithm (Consumer)}
    \begin{enumerate}
        \item Draw $\varepsilon_{ijt} \sim \text{Gumbel}(0,1)$ for all $i, j, t$
        \item Initialize $\bar{X}_{i0}$ (set mean or prior mean)
        \item For $t = 1, \ldots, T$:
            \begin{enumerate}
                \item Compute $U_{ijt} = \beta_i X_{jt}
                      - \gamma_i (X_{jt} - \bar{X}_{it})^2
                      + \kappa a_{jt} + \varepsilon_{ijt}$
                \item Set $j^* = \arg\max_j U_{ijt}$
                \item Update $\bar{X}_{it+1} = \frac{1}{t}\sum_{k \le t} x_{ijk^*}$
            \end{enumerate}
        \item Compute CCPs via log-sum-exp for numerical stability:
              $\log P_{ijt} = U_{ijt} - \log \sum_k e^{U_{ikt}}$
        \item Average over $S = 1{,}000$ simulation draws
    \end{enumerate}
\end{frame}

\begin{frame}{Firm Backward Induction}
    Terminal condition: $V_j(T+1) = 0$

    \vspace{6pt}
    For $t = T, T-1, \ldots, 1$:
    \[
        V_j(t) = \max_{a \in \{0,1\}}
        \Bigl[ s_{jt}(p - c) - \kappa a^2 + \delta \, V_j(t+1) \Bigr]
    \]

    Optimal policy: advertise if $V(\text{ad}) \ge V(\text{no ad})$:
    \[
        a_j^*(t) = \mathbf{1}\bigl[s_{jt}(p-c) - \kappa > s_{jt}(p-c) \bigr]
                 = \mathbf{1}[\kappa > 0]
    \]

    \textit{With the fixed binary spec the firm always advertises when $\kappa > 0$;
    the Markovian spec breaks this degeneracy by making cost state-dependent.}
\end{frame}

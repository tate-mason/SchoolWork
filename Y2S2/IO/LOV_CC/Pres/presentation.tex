\section{Motivation}
% ============================================================

\begin{frame}{Why Does Variety Matter?}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Canonical logit models ignore consumer preferences for variety, which are
              empirically important (e.g., McFadden 1974; Berry, Levinsohn,
              Pakes 1995)
        \item Variety-seeking and habit formation are two sides of the same coin:
              consumers either prefer to try new things or stick to what they know
        \item These preferences shape consumer switching behavior, which in turn
              shapes firms' incentives to target advertising and build market share
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{tcolorbox}[colback=red!5!white, colframe=UGARed, title=\textbf{Research Question}]
        How much do consumer tastes for variety and habit influence consumption dynamics and firms' advertising strategies in a dynamic market?
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{What This Paper Does}
  \begin{enumerate}
    \item \hl{Model:} Derive a discrete-choice demand system where the
          utility of a product depends on both deviation from the consumer's
          experienced mean $\bar{X}_{it}$ and product quality $X_{jt}$
    \vspace{4pt}
    \item \hl{Simulation:} Characterize consumer and firm behavior
          across preference regimes $(\beta_i, \gamma_i)$ and advertising
          specifications
    \vspace{4pt}
    \item \hl{Estimation (coming):} Bring model to Nielsen scanner data
          to recover structural parameters
    \vspace{4pt}
    \item \hl{Counterfactuals (coming):} Study targeting policy and
          welfare implications
  \end{enumerate}
  \vspace{6pt}
  \textbf{Today's talk:} Model + simulation results
\end{frame}

\section{Model}
% ============================================================

\begin{frame}{Consumer's Problem}
  Consumer $i$ chooses product $j$ in period $t$ to maximize:
  \begin{equation*}
    U_{ijt} = \underbrace{\beta_i X_{jt}}_{\text{quality gain}}
              + \underbrace{\gamma_i \Sigma_{ijt}^2}_{\text{variety component}}
              + \underbrace{\kappa \, a_{jt}}_{\text{advertising}}
              - \underbrace{\alpha \, p_{jt}}_{\text{price}}
              + \xi_j + \varepsilon_{ijt}
  \end{equation*}
  where $\Sigma_{ijt} = X_{jt} - \bar{X}_{it}$ and
  $\bar{X}_{it} = \frac{1}{t}\sum_{k=1}^{t} X_{ikt}$

  \vspace{8pt}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \hl{Key parameters:}
      \begin{itemize}
        \item $\beta_i > 0$: marginal utility of quality
        \item $\gamma_i > 0$: variety-seeking
        \item $\kappa > 0$: advertising shifter
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \hl{Choice probability (MNL):}
      \begin{equation*}
        P_{ijt} = \frac{e^{U_{ijt}}}{1 + \sum_{k} e^{U_{ikt}}}
      \end{equation*}
      Follows from $\varepsilon_{ijt} \sim \text{Gumbel}(0,1)$
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{State Variable: Experienced Mean}
  \begin{tcolorbox}[colback=red!5!white, colframe=UGARed, title=\textbf{Experienced Mean $\bar{X}_{it}$}]
    \[
      \bar{X}_{it} = \frac{1}{t} \sum_{k=1}^{t} x_{ikt}
    \]
    Updates each period as the consumer makes a new choice.
    This is the key state variable linking past behavior to current utility.
  \end{tcolorbox}

  \vspace{6pt}
  \hl{Two initialization strategies (both simulated today):}
  \begin{enumerate}
    \item \textbf{Set mean:} $\bar{X}_{i0} = \bar{J}$, the mean of the
          product set. Simple benchmark.
    \item \textbf{Initial state:} Simulate $T_{\text{prior}} = 5$ pre-sample
          periods from $J \in [1,2,3,4,5]$ to generate an experience-based
          prior. Similar to Bronnenberg, Dub\'e, Joo (2021).
  \end{enumerate}

  \vspace{1pt}
  \textit{Prior mean yields more heterogeneous behavior --- more on this shortly.}
\end{frame}

\begin{frame}{Firm's Problem}
  \hl{Firm $j$ solves a dynamic program:}
  \[
    V_j(t; \Sigma_{jt}, \lambda_t(\gamma_i)) = \max_{a \in \{0,1\}}
    \left\{ \pi_{jt}(a) + \beta_{df} \, \E\bigl[V_j(t{+}1;\cdot)\bigr] \right\}
  \]

  \vspace{4pt}
  \hl{Per-period profit:}
  \[
    \pi_{jt}(a) = s_{jt}(p - mc) - \mu \, a
  \]

  \vspace{4pt}
  \hl{Two advertising specifications:}
  \begin{itemize}
    \item \textbf{Fixed:} $a \in \{0, 1\}$ (binary on/off)
    \item \textbf{Heuristic} $a_t \propto (\bar{X}_{it} - X_{jt-1})^2$
          --- higher spending when product is far from consumer's mean
          --- work in progress: not ideal measure of drift
  \end{itemize}

  \vspace{4pt}
  Solved via backward induction; firm chooses $a$ to maximize discounted profits (to be completed)
\end{frame}

  \hl{Product 1:}
  \begin{itemize}
    \item Chosen by variety-seeking consumers ($\gamma_i$ high) who are drawn to the extreme initial state, 
    \item avoided by quality-sensitive consumers ($\beta_i$ high) who learn that it is low quality.
  \end{itemize}
  \hl{Product 3:}
  \begin{itemize}
    \item Chosen by the regime who is neither quality-sensitive nor variety-seeking (low $\beta_i$, low $\gamma_i$) who are content to stick to the mean.
    \item Small uptake by medium qualtiy-sensitive consumers (medium $\beta_i$, low $\gamma_i$), attempt to learn about quality but not variety-seeking enough to try other products.
  \end{itemize}
  \hl{Product 5:}
  \begin{itemize}
    \item Chosen by quality-sensitive consumers ($\beta_i$ high) who are drawn to the extreme initial state and learn that it is high quality.
    \item Avoided by variety seeking consumers ($\gamma_i$ high) and those who are neither ($\beta_i$ low, $\gamma_i$ low) who are content to stick to the mean.
    \item Highest probability of choice overall, consistent with idea of utility maximization
  \end{itemize}

\section{Next Steps}
% ============================================================

\begin{frame}{Road to Estimation}
  \begin{enumerate}
    \item \hl{Multiple consumers} \\
          Extend simulation to $I = 5$ heterogeneous consumers;
          aggregate to market shares

    \vspace{4pt}
    \item \hl{Heterogeneous $\gamma_i$} \\
          Draw $\gamma_i \sim N(0, \sigma_\gamma)$; simulate over
          $\sigma_\gamma \in \{0.5, 1.0, 1.5\}$

    \vspace{4pt}
    \item \hl{Consumer value function} \\
          Move from reduced-form simulation to solving the full
          dynamic discrete choice with continuation values

    \vspace{4pt}
    \item \hl{Nielsen scanner data} \\
          Match simulated moments to data;
          identify $(\beta_i, \gamma_i)$ using BLP instruments

    \vspace{4pt}
    \item \hl{Firm's Problem} \\
          Ad targeting policy; profit maximization; counterfactuals
  \end{enumerate}
\end{frame}

    I connect the love-of-variety demand literature to the dynamic
    targeting literature by making $(X_{jt}-\bar{X}_{it})$ the sufficient
    statistic for both consumer switching and firm ad decisions,
    as well as by introducing a novel way to generate consumer heterogeneity
    in this framework via the love-of-variety.
  \end{tcolorbox}

\begin{frame}{Summary}
  \begin{enumerate}
    \item \hl{Model:} Consumer utility depends on deviation from
          experienced mean; nests variety-seeking and habit formation

    \vspace{4pt}
    \item \hl{Set mean:} Flat choice probabilities --- good benchmark,
          insufficient for data

    \vspace{4pt}
    \item \hl{Prior mean:} Rich behavioral dynamics; regime differences
          generate cross-sectional sorting consistent with scanner data

    \vspace{4pt}
    \item \hl{Firm:} Markovian advertising incentive rises when
          consumer drifts; generates time-varying promotional intensity

    \vspace{4pt}
    \item \hl{Next:} Multiple consumers, structural estimation,
          Nielsen data
  \end{enumerate}

\begin{frame}{Simulation Algorithm (Consumer)}
  \begin{enumerate}
    \item Draw $\varepsilon_{ijt} \sim \text{Gumbel}(0,1)$ for all $i, j, t$
    \item Initialize $\bar{X}_{i0}$ (set mean or prior mean)
    \item For $t = 1, \ldots, T$:
      \begin{enumerate}
        \item Compute $U_{ijt} = \beta_i X_{jt}
              - \gamma_i (X_{jt} - \bar{X}_{it})^2
              + \kappa a_{jt} + \varepsilon_{ijt}$
        \item Set $j^* = \arg\max_j U_{ijt}$
        \item Update $\bar{X}_{it+1} = \frac{1}{t}\sum_{k \le t} x_{ijk^*}$
      \end{enumerate}
    \item Compute CCPs via log-sum-exp for numerical stability:
          $\log P_{ijt} = U_{ijt} - \log \sum_k e^{U_{ikt}}$
    \item Average over $S = 1{,}000$ simulation draws
  \end{enumerate}
\end{frame}

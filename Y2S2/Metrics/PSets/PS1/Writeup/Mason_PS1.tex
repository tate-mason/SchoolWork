  \subquestion{Show analytically why the MLE and MoM approaches yield identical parameter estimates to OLS}
  \begin{solution}
    Let's first derive $\hat{\beta}$ via MLE:
    \begin{gather*}
      f(Y_i|X_i;\beta,\sigma) = \frac{1}{\sqrt{2\pi\sigma^2}}\cdot\exp(-\frac{(Y_i - X_i'\beta)^2}{2\sigma^2}) \\
      \mathcal{l} = -\frac{n}{2}\ln(2\pi) - \frac{n}{2}\ln(\sigma^2) - \frac{1}{2\sigma^2}(Y_i - X_i'\beta)^2 \\
      \frac{\partial\mathcal{l}}{\partial\beta} = -2\sum\limits_{i=1}^{N}(X_i'Y_i) + 2N\beta\sum\limits_{i=1}^{N}(X_i'X_i) = 0 \\
      N\beta = \sum\limits_{i=1}^{N}(X_i'X_i)^{-1}(X_i'Y_i) \\
      \hat{\beta}_{MLE} = \frac{1}{N}\sum\limits_{i=1}^{N}(X_i'X_i)^{-1}(X_i'Y_i) = \hat{\beta}_{OLS}
    \end{gather*}
    As can be seen, we end up at the same sample estimate for $\hat{\beta}$ which will converge in probability to the true $\beta$ as $N \rightarrow \infty$

    For MoM:
    \begin{gather*}
      \frac{1}{N}\sum\limits_{i=1}^{N}(X_i'e_i) = 0; \ e_i = Y_i - X_i\beta \\
      \frac{1}{N}\sum\limits_{i=1}^{N}(X_i'(Y_i - X_i\beta)) = 0 \\
      \frac{1}{N}\sum\limits_{i=1}^{N}(X_i'Y_i - (X_i'X_i)\beta) = 0 \\
      \frac{1}{N}\sum\limits_{i=1}^{N}(X_i'Y_i) - \frac{1}{N}\sum\limits_{i=1}^{N}(X_i'X_i)\beta = 0 \\
      \hat{\beta}_{MoM}= \frac{1}{N}\sum\limits_{i=1}^{N}(X_i'X_i)^{-1}(X_i'Y_i) = \hat{\beta}_{OLS}
    \end{gather*}
    Here, using assumptions on the residual, we can generate the same $\hat{\beta}$ in both MoM and OLS.
  \end{solution}

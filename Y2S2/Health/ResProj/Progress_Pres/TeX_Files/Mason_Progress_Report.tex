\section{Motivation \& Research Question}
% ============================================================

\begin{frame}{Motivation}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Title IX (1972) dramatically expanded female athletics, but \red{role-model effects} remain understudied.
        \vspace{0.4em}
        \item WNBA founded 1997 — first major professional women's basketball league in the US.
        \vspace{0.4em}
        \item Franchises entered states \textbf{staggered across 1997--2018}, creating quasi-experimental variation.
        \vspace{0.4em}
        \item \navy{Question:} Does local exposure to a professional women's team \emph{increase} girls' participation in high school sports?
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Why it matters}
        \small
        \begin{itemize}
          \item Physical/mental health benefits of youth sports
          \item Long-run labour market returns to athletic participation
          \item Informing policy on sports funding and media representation
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Empirical Strategy}
% ============================================================

\begin{frame}{Baseline TWFE Specification}
  \begin{block}{Estimating equation}
    \small
    \[
      Y_{it} = \beta_0 + \beta_1 \mathbbm{1}\{\text{WNBA}_{it}\} + \epsilon_{it}
    \]
  \end{block}
  \begin{itemize}
    \item $Y_{it}$: number of girls participating in a given sport in state $i$ and year $t$.
    \item $\mathbbm{1}\{\text{WNBA}_{it}\}$: indicator for whether state $i$ has a WNBA team in year $t$.
    \item $\epsilon_{it}$: error term.
    \item Future versions will include state and year fixed effects, as well as covariates (e.g., population, median income).
  \end{itemize}
  \textit{This specification is biased with staggered treatment timing, so I correct with Callaway \& Sant'Anna (2021) method.}
  
\end{frame}


\begin{frame}{Identification Strategy}
  \textbf{Callaway \& Sant'Anna (2021)} staggered difference-in-differences:

  \vspace{0.6em}
  \begin{itemize}
    \item Define \textbf{group} $g$ = first year state $i$ gets a WNBA team (\textit{or} $g = 0$ if never treated).
    \vspace{0.4em}
    \item Estimate \textbf{group-time average treatment effects}:
      \[
        \attg = \mathbb{E}[Y_{it}(g) - Y_{it}(0) \mid G_i = g]
      \]
    \item \textbf{Control group:} never-treated states only (33 states).
    \item Allows for staggered timing whereas baseline TWFE is biased
    \item Standard errors clustered by state; 95\% uniform confidence bands.
  \end{itemize}

  \vspace{0.6em}
  \begin{block}{Key assumption}
    \small Parallel trends conditional on covariates: in the absence of a WNBA team, treated and never-treated states would have evolved similarly. We test this via pre-treatment event-study coefficients.
  \end{block}
\end{frame}

\section{Conclusion}
% ============================================================


\begin{frame}{Priority Items \& Limitations}
  \begin{itemize}
    \item NFHS data digitised from annual PDF reports — potential for transcription errors. 
    \item WNBA treatment defined at state level; may not capture local variation in media exposure or fan engagement.
    \item Baseline TWFE estimates are biased with staggered treatment timing; C\&S method addresses this but relies on parallel trends assumption.
    \item Event-study coefficients have wide confidence intervals, especially for non-basketball sports, limiting precision of inference.
    \item Add state and year covariates (e.g., population, median income) to improve precision and test robustness.
  \end{itemize}
\end{frame}
\begin{frame}{Summary \& Implications}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Main findings}
      \begin{itemize}
        \item Pre-trends pass visual inspection across all sports.
        \item WNBA entry associated with a modest, but significant increase in basketball participation (approx. \ 2000 additional girls)
        \item No signficant effects on soccer or cross country; some suggestive evidence for track \& field seeing a boost, but statistical significance is borderline.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Economic implications}
        \small
          Role-model effects can boost youth sports participation, with potential long-term health and economic benefits.
      \end{block}
      \vspace{0.4em}
      \begin{block}{Next steps}
        \small
        \begin{itemize}
          \item Synthetic Difference-in-Differences
          \item Explore substitution vs.\ complementarity across sports (e.g., does WNBA entry boost basketball but reduce soccer?)
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

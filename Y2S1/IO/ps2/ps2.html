# Creation of Investment Variable with 10 per depreciation rate
data <- data %>%
  group_by(id) %>%
  mutate(
    dep = .1,
    k_next = lead(K, 1),
    i =  k_next - ((1 - dep) * K)
  ) %>%
  group_by(industry) %>%
  mutate(
    log_i = log(i)
  ) %>%
  ungroup()

\section{Motivation \& Facts}
% ============================================================

\begin{frame}{Question \& Motivation}
  \begin{itemize}
    \item \hl{Question:} Does Amazon's vertical integration structure harm competition on its marketplace platform? 
    \item Amazon is able to act as both a retailer and a platform for third-party sellers (3PS)
    \item What is the effect of this dual role as both competitor and marketplace operator on consumer welfare and competition?
  \end{itemize}
\end{frame}

\begin{frame}{Background Facts}
  \begin{itemize}
    \item Amazon's share of total e-commerce sales was 40.4\% in 2024
    \item Amazon experiences higher demand for its own products than 3PS do for theirs
    \begin{itemize}
      \item 3PS sell 3 products per week
      \item Amazon brands sell 221 per week
      \item Amazon is only seller of $\approx$ half of the product it offers
    \end{itemize}
    \item 3PS have two modes of fulfillment:
    \begin{itemize}
      \item Fulfilled by Amazon: Amazon fulfills delivery for a fee
      \item Fulfilled by Merchant: Cost of storage and delivery
    \end{itemize}
    \item \hl{What to do?}
  \end{itemize}
\end{frame}

\begin{frame}{Contribution}
  \begin{itemize}
    \item Estimate a counterfactual in which Amazon is no longer a competitor
    \item Extend counterfactual to have a 3PS take Amazon's place
    \item Estimates demand, indirect utility, find bounds on fixed costs, structural separation
  \end{itemize}
\end{frame}

\begin{frame}{Paper’s Story / Key Findings}
  \begin{itemize}
    \item Counterfactual:
    \begin{itemize}
      \item \hl{No Re-entry:} Consumer welfare falls 18.6\%, price increases by 1.7 pp., 20.6\% profit increase for 3PS
      \item \hl{Re-entry:} Consumer welfare still falls, but up to 4.4 pp. less than in the above case
    \end{itemize}
    \item Amazon as a seller appears to contribute value, thus implying the need for thought on the part of regulators
  \end{itemize}
\end{frame}

\section{Model}

\begin{frame}{Model Setup}
  \begin{itemize}
    \item Discrete choice demand model
    \item Products indexed by $j = 1, \ldots, J$ sold by sellers $r = 1, \ldots, R$ over weeks $t = 1, \ldots, T$ and months $m = 1, \ldots, M$
    \item 3PS pay fee $\phi$ dependent upon upon charged price, and additional fixed $\tau$ if using FBA
    \item FBM seller $r$ has a product $j \in \mathcal{J}_{rt}$ with price $p_{rt}$
    \item FBA seller $r$ has product $j \in \tilde{\mathcal{J}}_{rt}$ with price $p_{rt}$
    \item Sellers make two choices each week: set of products to offer and prices
    \item Set $\textbf{\mathcal{J}}_{rt} = \{\mathcal{J}_{rt}, \tilde{\mathcal{J}}_{rt}\}$ defines all products offered by seller $r$ in week $t$
  \end{itemize}
\end{frame}

\begin{frame}{Demand}
  \begin{itemize}
    \item Indirect utility of consumer $i$ for product $j$ in week $t$ is:
    \beq{eq:utility}
    U_{ijt} = \beta\textbf{X}_{jrt} - \alpha_ip_{jrt} + \gamma_j + \gamma_m + \xi_{jrt} + \epsilon_{ijrt}
    \eeq
    \item $\X_{jrt}$: observed seller and product characteristics
    \item $\gamma_j$: product fixed effects
    \item $\gamma_m$: month fixed effects
    \item $\xi_{jrt}$: unobserved product characteristics
    \item $\epsilon_{ijrt}$: i.i.d. extreme value error term s.t. outside option $U_{i0t} = \epsilon_{i0t}$
  \end{itemize}
\end{frame}

\begin{frame}{Demand Cont.}
  \begin{itemize}
    \item Market share prediction is:
    \beq{eq:share}
      s_{jrt}
      = \int
      \frac{
        \exp\{\delta_{jrt} + \sigma \mu_i\}
      }{
        1
        + \sum_{r = 1}^R \sum_{j \in \mathcal{J}_{rt}} \exp\{\delta_{jrt}\}
        + \sum_{j \in \mathcal{J}_{at}} \exp\{\delta_{jat} + \sigma \mu_i\}
      } \, dF(\mu)
    \eeq
    \item $\delta_{jrt} = \beta\X_{jrt} - \alpha p_{jrt} + \gamma_j + \gamma_m + \xi_{jrt}$
    \item $\sigma\mu_i$: random coefficient on price
  \end{itemize}
\end{frame}

\begin{frame}{Variable Profit}
  \begin{itemize}
    \item Marginal cost for 3PS differs by fulfillment method:
    \begin{itemize}
      \item FBM: $c_{jrt}$
      \item FBA: $\tilde{c}_{jrt}$
    \end{itemize}
    \item Firm variable profits:
    \begin{align}
      \pi_{rt}(\mathcal{J}_{rt}, X_t)
      &= \mathcal{M}_t \Bigg(
        \sum_{j\in\mathcal{J}_{rt}} s_{jrt}(p_t, X_t)\big[(1-\phi)p_{jrt} - c_{jrt}\big] \\
      &\qquad\quad
        + \sum_{j\in\tilde{\mathcal{J}}_{rt}} s_{jrt}(p_t, X_t)\big[(1-\phi)p_{jrt} - \tilde{c}_{jrt} - \tau\big]
      \Bigg)
    \end{align}
  \end{itemize}
\end{frame}

\begin{frame}{Amazon Profit}
  \begin{itemize}
    \item Amazon profit:
    \begin{align}
      \pi_{at}(\mathcal{J}_{at}, X_t)
      &= \mathcal{M}_t \Bigg(
        \sum_{j\in\mathcal{J}_{at}} s_{jat}(p_t, X_t)\big[p_{jat} - c_{jat}\big] \\
      &\qquad\quad
        + \sum_{r\ne a}\sum_{j\in\mathcal{J}_{rt}} s_{jrt}(p_t, X_t)\,\phi\,p_{jrt}
        + \sum_{r\ne a} \tau \cdot \mathbf{1}\{j\in\tilde{\mathcal{J}}_{rt}\}
      \Bigg)
    \end{align}
  \end{itemize}
\end{frame}

\begin{frame}{Costs}
  \begin{itemize}
    \item Marginal Costs are are derived from supply functions:
    \beq{eq:marginal_cost}
      c_{jrt} = (\delta_1 + \delta_2 \mathbf{1}_{r=a})\mathbf{1}_{FBA} + \zeta_j + \zeta_t + \omega_{jrt}
    \eeq
    \item $\zeta_j$: product fixed effects, $\zeta_t$: week fixed effects, $\omega_{jrt}$: error
    \item Fixed costs
    \beq{eq:fixed_cost}
      f_{jrt} = \begin{cases}
        \theta_a + \psi_{jat}, & \text{if Amazon} \\
        \theta + \psi_{jrt}, & \text{if 3PS using FBM} \\
        \tilde{\theta} + \tilde{\psi}_{jrt}, & \text{if 3PS using FBA}
      \end{cases}
    \eeq
    \item $\psi$: product-seller specific unobserved fixed cost component
    \item $\tilde{\theta}$ takes into account opportunity cost of using FBA vs. other methods
    \item $\theta_a$ represents inventory costs and opportunity cost of selling for Amazon
    \item $\theta$ represents fixed costs for 3PS using FBM, including storage, refilling stock, etc.
  \end{itemize}
\end{frame}

\section{Estimation Idea}
% ============================================================

\begin{frame}{Estimation Overview}
  \begin{itemize}
    \item Estimate demand via GMM using BLP framework, recover $(\beta, \alpha, \gamma, \sigma)$
    \begin{itemize}
      \item Instruments: BLP and product characteristics of other products
    \end{itemize}
    \item Recover marginal costs via first order conditions of profit maximization
    \item Back out bounds on fixed costs via entry/exit decisions (revealed preferences)
  \end{itemize}
\end{frame}

\section{Conclusion}
% ============================================================

\begin{frame}{Takeaways}
  \begin{itemize}
    \item Amazon's dual role as both a retailer and a platform operator has significant effects on competition and consumer welfare
    \item Counterfactual analysis shows that removing Amazon as a seller leads to a decrease in consumer welfare, even with potential re-entry by other sellers
    \item The presence of Amazon as a seller appears to contribute value to the marketplace, suggesting that regulators should carefully consider the implications of vertical integration in e-commerce platforms
  \end{itemize}
\end{frame}

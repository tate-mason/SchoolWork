\begin{frame}{Question \& Motivation}
  \begin{itemize}
    \item \hl{Question:} Does Amazon's vertical integration structure harm competition on its marketplace platform? 
    \item Amazon is able to act as both a retailer and a platform for third-party sellers (3PS)
    \item What is the effect of this dual role as both competitor and marketplace operator on consumer welfare and competition?
  \end{itemize}
\end{frame}

\begin{frame}{Background Facts}
  \begin{itemize}
    \item Amazon's share of US e-commerce sales increased from 27\% in 2015 to 41\% in 2022
    \item 3PS account for over 50\% of units sold on Amazon
    \item Amazon Retail (AR) has consistently been the top-selling seller on its own platform
    \item AR can use non-public data from 3PS to inform its own pricing and product decisions
    \item Concerns about potential anti-competitive behavior have led to regulatory scrutiny in the US and EU
  \end{itemize}
\end{frame}

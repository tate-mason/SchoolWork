\section{Paper Overview \& Contribution}

\begin{frame}{Overview of Paper}
  \begin{itemize}
    \item $\frac{1}{5}$ of non-elderly uninsured in 2005.
    \item Due to Medicare eligibility at 65, $ < 1\%$ of elderly are uninsured.
    \item \hl{Research question}: How does gaining near-universal insurance at 65 affect health care utilization?
    \item \hl{Identification}: RD exploiting sharp age cutoff.
    \item \hl{Data}: Hospital discharge records from CA, FL, NY and NHIS survey data.
  \end{itemize}
\end{frame}

\begin{frame}{Contribution (Short Discussion)}
  \begin{itemize}
    \item \hl{What’s new}: Broader range of outcomes than previous work, quantify differences after receiving Medicare between groups 
    \item \hl{Why it matters}: Inform policy debates about expanding insurance coverage.
  \end{itemize}
\end{frame}

\section{Story \& Key Findings}

\begin{frame}{Paper’s Story / Key Findings}
  \hl{Key qualitative takeaways}
  \begin{itemize}
    \item Increase in overall coverage, with lessening coverage gaps.
    \item Increase in multiple coverage, especially for highly educated, non-minorities.
    \item Increase in medical care utilization, though effects differ by group
  \end{itemize}
  \hl{Key quantitative takeaways}
  \begin{itemize}
    \item Medicare coverage increases 60 pp at age 65.
    \item Probability of any insurance coverage increases by 9.5 pp
    \item Probability of multiple coverage increases by 44.1 pp
  \end{itemize}
\end{frame}

\section{Empirical Strategy}

\begin{frame}{Reduced-Form Specification}
  \beq{eq:main}
    y_{ija} = X_{ija}\alpha + f_j(a;\beta) + \Sigma_k C_{ija}^k\delta^k + u_{ija}
  \eeq
  \begin{itemize}
    \item $y_{ija}$: outcome for individual $i$ in group $j$ at age $a$.
    \item $X_{ija}$: individual, regional, etc. controls 
    \item $f_j(a;\beta)$: smoothing function based on age profile
    \item $C_{ija}^k$: characteristics of insurance coverage, can include coverage indicator or other insurance covariates
  \end{itemize}
\end{frame}

\begin{frame}{Coverage Functions}
  \beq{eq:any coverage}
    C_{ija}^1 = X_{ija}\beta_j^1 + g_j^1(a) + D_a\pi_j^1 + v_{ija}^1
  \eeq
  \beq{eq:generous coverage}
    C_{ija}^2 = X_{ija}\beta_j^2 + g_j^2(a) + D_a\pi_j^2 + v_{ija}^2
  \eeq
  \begin{itemize}
    \item $\beta_j^1, \beta_j^2$: group-specific coefficients.
    \item $g_j^1(a), g_j^2(a)$: smoother for group-specific age profiles.
    \item $D_a$: indicator for age 65+.
  \end{itemize}
\end{frame}

\begin{frame}{Final Reduced Form}
  \beq{eq:final}
    y_{ija} = X_{ija}(\alpha_j + \beta_j^1\delta_j^1 + \beta_j^2\delta_j^2) + h_j(a) + D_a\pi_j^y + \nu_{ija}^y
  \eeq
  \begin{itemize}
    \item $h_j(a)$: reduced form age profile.
    \item $\pi_j^y$: aggregated effect of Medicare eligibility on outcome $y$.
    \item $\nu_{ija}^y$: new error term.
  \end{itemize}
  Assuming profiles $f_j(a), g_j^1(a), g_j^2(a)$ are all continuous at 65, any discontinuity in $y$ at 65 is due to Medicare eligibility.
\end{frame}

\begin{frame}{Overcoming Pitfalls}
  \begin{itemize}
    \item \hl{Lack of Data} Unable to clearly observe information on insurance packages of each individual
    \item \textcolor{blue}{Solution}: Use at least two measures of coverage to proxy for insurance generosity.
    \item \hl{Endogeneity of Coverage} Insurance coverage is all that matters, meaning the effect will differ by group.
    \item \textcolor{blue}{Solution}: Estimate separate age profiles and coverage functions by group.
    \beq{eq:disconinuity est}
      \pi_j^y = \delta^0 + \delta^1\pi_j^1 + \delta^2\pi_j^2 + e_j
    \eeq
  \end{itemize}
\end{frame}

\section{Conclusion}

\begin{frame}{Conclusion}
  \hl{What we learned}:
  \begin{itemize}
    \item Medicare eligibility at 65 leads to large increases in healthcare utilization, especially for minorities
    \item Medicare leads to increased insurance coverage, especially multiple coverage
    \item admissions for discretionary procedures increase more than emergency admissions across all groups, though is particularly pronounced for Hispanics
  \end{itemize}
\end{frame}

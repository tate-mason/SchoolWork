\section{Model Intuition}
% ============================================================

\begin{frame}{Model Basics}
  \begin{itemize}
    \item \hl{Setup:} Locations differ in mean wages and match quality; moving is costly.
    \item \hl{Information:} Current location wage known; other locations' matches known only by visiting (experience good).
    \item \hl{Decision:} Maximize expected lifetime income net of moving costs.
  \end{itemize}
  \begin{center}

\section{Estimation Idea}
% ============================================================

\begin{frame}{Computational Challenge \texorpdfstring{$\rightarrow$}{->} Limited-History Approximation}
  \begin{itemize}
    \item Full state space explodes with locations and match histories.
    \item \hl{Idea:} Keep only the most recent $M$ locations for wages/matches.
    \item Preserves behavior seen in the data while making DP feasible.
  \end{itemize}
  \begin{center}
\begin{tikzpicture}
  \node[draw, rounded corners, align=left, inner sep=6pt] (full) {Full state (infeasible):\\
  $\big(\text{all past locations},\ \{\text{all }w,\xi\}\big)$};
  \node[right=3.8cm of full, draw, rounded corners, align=left, inner sep=6pt, fill=UGAGray!10] (limited) {Limited history ($M$ recent):\\
  $\big((\ell_0,\ell_1),\ (w_{\ell_0},w_{\ell_1}),\ (\xi_{\ell_0},\xi_{\ell_1})\big)$};
  \draw[-{Latex[length=3mm]}] (full) -- node[above]{truncate to last $M$} (limited);
\end{tikzpicture}
\end{center}
  % FORMAL STATE TRANSITIONS/LIKELIHOOD IN BACKUP.
\end{frame}

\begin{frame}{Wage Components (What Matters for Moves)}
  \begin{itemize}
    \item Mean wage by location; permanent match component; person FE; transitory shocks.
    \item Migration choice depends on \textit{differences} in expected income and moving cost, not levels.
  \end{itemize}
  \begin{block}{Wage decomposition (relevant pieces in bold)}
\begin{equation*}
\underbrace{w_{ij}(a,t)}_{\text{observed}}=
\underbrace{\mu_j}_{\text{mean by location}} + \ \underbrace{\boldsymbol{\upsilon_{ij}}}_{\textbf{match}} + G(X_i,a,t) + \eta_i + \varepsilon_{ijt}
\end{equation*}
\end{block}
\begin{itemize}
  \item Migration choice depends on $\boldsymbol{\mu_j + \upsilon_{ij}}$ and moving cost $\Delta(\ell_0,j)$.
\end{itemize}
\end{frame}

\section{Results}
% ============================================================

\begin{frame}{What Drives Migration? (Headlines)}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\linewidth}
      \begin{itemize}
        \item \hl{Expected income matters}: moves respond to geographic mean wages and bad current matches.
        \item \hl{Costs matter}: distance penalty; strong home premium; cheaper to return.
        \item \hl{Population size} of destination reduces cost (network/proximity channels).
      \end{itemize}
    \end{column}
    \begin{column}{0.48\linewidth}

    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Economic Magnitudes}
  \begin{itemize}
    \item Moving cost equivalent $\approx$ \$300k (PV terms, order of magnitude).
    \item Good match + high-mean state \texorpdfstring{$\Rightarrow$}{->} $\approx\$$22k / year gain (1989).
  \end{itemize}
\end{frame}

  \begin{itemize}
    \item Model reproduces repeat/return migration shares qualitatively.
    \item Negative association: higher realized income this period $\Rightarrow$ lower out-migration probability.
  \end{itemize}
\end{frame}

% ============================================================
\section{Interpretation \& Policy}
% ============================================================

\begin{frame}{What We Learn}
  \begin{itemize}
    \item Migration resembles optimal search over locations.
    \item Both mean differences and idiosyncratic match quality drive moves.
    \item Structural approach enables policy counterfactuals.
  \end{itemize}
\end{frame}

\begin{frame}{Policy Angles \& Extensions}
  \begin{itemize}
    \item Local labor demand shocks \& wage equalization dynamics.
    \item Subsidies or relocation assistance: who moves more?
    \item Household joint decisions; welfare benefit differentials; retirement migration.
  \end{itemize}
\end{frame}

% ============================================================
\section{Conclusion}
% ============================================================

\begin{frame}{Takeaways}
  \begin{itemize}
    \item Expected income plays a sizable role in interstate migration.
    \item High moving costs + payoff shocks reconcile low average move rates with meaningful gains.
    \item Model provides a tractable template for counterfactual policy analysis.
  \end{itemize}
\end{frame}

\begin{frame}{Value Function \& Choice Probabilities \textit{(Backup)}}
  \begin{block}{Bellman equation}
\begin{align*}
V(x,\boldsymbol{\zeta}) &= \max_{j\in \mathcal{J}} \{\, v(x,j) + \zeta_j \,\},\\
v(x,j) &= u(x,j) + \beta \mathbb{E}_{x'}[\, \bar V(x') \mid x,j\,], \quad \bar V(x)=\mathbb{E}_{\zeta}[V(x,\zeta)].
\end{align*}
\end{block}
\begin{block}{Extreme value shocks $\Rightarrow$ logit choice probabilities}
\begin{equation*}
\rho(x,j)=\frac{\exp(v(x,j))}{\sum_{k\in\mathcal{J}} \exp(v(x,k))},\qquad
\exp(\bar V(x))=e^{\bar\gamma}\sum_{k}\exp(v(x,k)).
\end{equation*}
\end{block}
\end{frame}

\begin{frame}{Limited-History State Transitions \textit{(Backup)}}
  \begin{block}{State and transitions (with $M=2$ recent locations)}
State: $x=(\ell_0,\ell_1,\ \omega_{\ell_0}^{(w)},\omega_{\ell_1}^{(w)},\ \omega_{\ell_0}^{(\xi)},\omega_{\ell_1}^{(\xi)},\ a).$\\[0.3em]
If stay: $x'=(\ell_0,\ell_1,\ \cdot,\cdot,\cdot,\cdot,\ a+1).$\\
If move to previous $\ell_1$: swap $(\ell_0,\ell_1)$ and their components, $a\leftarrow a+1.$\\
If move to new $j$: shift $(\ell_0\!\leftarrow\! j,\ \ell_1\!\leftarrow\! \ell_0)$ and draw new $(\omega_j^{(w)},\omega_j^{(\xi)})$.
\end{block}
\end{frame}

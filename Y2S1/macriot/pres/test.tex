\section{Motivation \& Facts}
% ============================================================

\begin{frame}{Question \& Motivation}
  \begin{itemize}
    \item \hl{Question:} Do expected income differences drive interstate migration?
    \item Migration is common for the young; repeat and return moves are pervasive.
    \item Policy relevance: local shocks, subsidies, welfare differentials.
  \end{itemize}
\end{frame}

\begin{frame}{Models of Migration}
  \begin{itemize}
    \item Migration models are used to estimate how and why agents move within an economy
    \item Typically used to look at isolated migration events, which may be too shallow of a viewpoint
  \end{itemize}
\end{frame}

\begin{frame}{Contribution}
  \begin{itemize}
    \item \hl{What’s new}: Develops a tractable econometric model focusing on expected income's influence on migration which allows for dynamic decision making
    \item \hl{Why it matters}: Adds possibility of both repeat and return migration, creating a context for this model to solve a job-search problem
    \item Using data from NLSY, gain an understanding of the motives behind and patterns exhibited by movers in the framework of this model
  \end{itemize}
\end{frame}

\begin{frame}{Paper’s Story / Key Findings}
  \begin{itemize}
    \item The paper aims to look at white men with high school education, utilizing the NLSY to track agents through time
    \item Expected income prospects are a major contributor to interstate migration decisions
    \item Further, there is evidence of geographic factors taking place once income realization is met, and the current location is found to be suboptimal
    \item \hl{Key quantitative takeaways}
    \begin{itemize}
      \item Value of moving from a bad location to a good one equates to a \$17,000 per year
      \item Moving is mostly decided by fit of draw: if good, they stay; if bad: they try again, incurring a cost
    \end{itemize}
  \end{itemize}
\end{frame}

\section{Model Intuition}
% ============================================================

\begin{frame}{Model Assumptions}
  \begin{itemize}
    \item Assume wages are a local product of skill bundles
    \item $J$ locations
    \item $y_{ij}$ is individual $i$'s income in location $j$ with a known distribution
    \item Migration occurs to maximize expected discounted value of lifetime utility
    \item Ignore assets for this case, only seeking to maximize expected lifetime income, net of moving costs
  \end{itemize}
\end{frame}

\begin{frame}{Value Function}
  \begin{equation}
    V(x, \zeta) = \max_j(v(x,j) + \zeta_j)
  \end{equation}
  where,
  \begin{gather*}
    v(x,j) = u(x,j) + \beta\sum_{x'}p(x'|x,j)\bar{v}(x') \\
    \shortintertext{and} \\
    \bar{v}(x) = \mathbb{E}_{\zeta}V(x,\zeta)
  \end{gather*}
\end{frame}

\begin{frame}{Value Function, cont.}
  Further,
  \begin{gather*}
    \exp(\bar{v}(x)) = \exp(\bar{\gamma})\sum\limits_{j=1}^J\exp(v(x,k)), \\
    \shortintertext{and} \\
    \rho(x,j) = \exp(\bar{\gamma}+v(x,j)-\bar{v}(x))
  \end{gather*}
  where $\bar{\gamma}$ is the Euler constant, $\rho(x,j)$ is the probability of choosing $j$ when in state $x$
\end{frame}

\begin{frame}{Wages}
  Wage of agent $i$ in location $j$ at age $a$ during time $t$ is as follows,
  \begin{equation}
    w_{ij}(a) = \mu_j + v_{ij} + G(X_i, a, t) + \eta_i + \epsilon_{ij}(a)
  \end{equation}
  \begin{itemize}
  \item Quality of match governed by the difference between $\mu_j + v_{ij}$ and $\mu_k + v_{ik}$, the current quality versus the prospective quality of another location
  \item Match quality is known for current and, where possible, most recent location
  \item Prospects of all other locations are unknown
  \end{itemize}
\end{frame}

\begin{frame}{State Variables \& Flow Payoffs}
  \begin{itemize}
    \item $l = (l^0, l^1, ..., l^{M-1})$ be an $M$ vector containing the sequence of previous locations, beginning with the current location
    \item $\omega$ is an $M$ vector containing wage and utility information at these locations
    \item $l,\omega, a \in x$, the state vector
  \end{itemize}
  \begin{equation}
    \tilde{u}_h(x,j) = u_h(x,j) + \zeta_j,
  \end{equation}
  where,
  \begin{gather*}
    \begin{split}
      u_h(x,j) &= \alpha_0w(l^0,\omega) + \sum\limits_{k=1}^K\alpha_kY_k(l^0)+\alpha^H\Chi(l^0=h) \\
              &+ \xi(l^0,\omega) - \Delta_{\tau}(x,j)
    \end{split}
  \end{gather*}
\end{frame}

\begin{frame}{Moving Costs}
  \begin{itemize}
    \item $D(l^0, j)$ represents distance from current location to some other location $j$
    \item $\mathbb{A}(l^0)$ set of locations adjacent to $l^0$
  \end{itemize}
  Thus, define moving cost as
  \begin{equation}
    \begin{split}
      \Delta_{\tau}(x,j) &= (\gamma_{0,\tau} + \gamma_1D(l^0, j) - \gamma_2\Chi(j\in\mathbb{A}(l^0)) \\
                 &- \gamma_3\Chi(j=l^1) + \gamma_4a - \gamma_5n_j)\Chi(j\ne l^0)
    \end{split}
  \end{equation}
\end{frame}

\section{Estimation Idea}
% ============================================================

\begin{frame}{Likelihoood Function}
  \begin{equation}
    \Lambda(\theta) = \sum\limits_{n=1}^N\log(\sum\limits_{\tau=1}^K\pi_{\tau}L_i(\theta_{\tau}))
  \end{equation}
  \begin{itemize}
    \item Observed income
    \item Location choice
  \end{itemize}
\end{frame}

\begin{frame}{Likelihoods}
  Let $\psi_{it}(\omega_i, \theta)$ correspond to likelihood of observed income of agent $i$ in period $t$
  \begin{equation}
    \begin{split}
      \psi_{it} &(\omega_i,\theta_{\tau})  \\
      &=\frac{1}{\sigma_{\epsilon}(\omega_{\epsilon}^i)}\phi(\frac{w_{it} - \mu_{l^0(i,t)} 0 G(X_i, a_{it},\theta_{\tau}) - v(\omega_v^i(\kappa_{it}^0)) - \eta(\omega_{\eta}^i)}{\sigma_{\epsilon}(\omega_{\epsilon}^i)})
    \end{split}
  \end{equation}
  Let $\lambda_{it}(\omega^i, \theta_{\tau})$ be the likelihood of the destination chosen by agent $i$ in period $t$
  \begin{equation}
    \begin{split}
      \lambda_{it}(\omega^i, \theta_t) =& \rho_{h(i)}(l(i,t), \omega_v^i(\kappa_{it}^0), \omega_v^i(\kappa_{it}^1), \omega_{\xi}^i(\kappa_{it}^0), \\
                   & \omega_{\xi}^i(\kappa_{it}^1), a_{it}, l^0(i, t+1), \theta_{\tau})
    \end{split}
  \end{equation}
\end{frame}

\begin{frame}{Likelihoods, cont.}
  Likelihood of individual history for a person of type $\tau$ is
  \begin{equation}
    \begin{split}
      L_i(\theta_{\tau}) = &\frac{1}{n_{\eta}n_{\epsilon}n_{\xi}(n_v)^{N_i}} \\
      & \times \sum\limits_{\omega^i\in\Omega(N_i)}(\Pi\limits_{t=1}^{T_i}\psi_{it}(\omega^i,\theta_{\tau})\lambda_{it}(\omega^i,\theta_{\tau}))
    \end{split}
  \end{equation}
\end{frame}

\begin{frame}{Nonparametric Identification of Choice Probabilities}
  \begin{itemize}
  \item Wage matching cannot be observed directly
  \item We observe a match with error
  \item Dealt with via likelihood integrating over the distributions  of the errors
  \item Still, heavily reliant on assumptions
  \end{itemize}
  Wage residual can be defined as such
  \begin{equation}
    y_{it} = \eta_i + v_{ij(t)}+\epsilon_{it}
  \end{equation}
\end{frame}

\begin{frame}{DGP}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\linewidth}
      \begin{tikzpicture}[node distance=2.5m, auto, >=Stealth]
        \node (y1) {$y_1 = \eta + v_1 + \varepsilon_1$};

        \draw[->] (y1) -- (y2move) node[midway, above left] {$\rho(\nu)$};
        \node[right=of y1, yshift=-1.2cm] (y2stay) {$y_2 = \eta + \tilde{v}^S + \varepsilon_2$};

      \end{tikzpicture}
    \end{column}
    \begin{column}{0.48\linewidth}
      \begin{itemize}
        \item with probability $\rho(v)$, move and get observed wage $y_2 = \eta + v' + \varepsilon_2$
        \item with probability $1-\rho(v)$, stay and get $y_2 = \eta + \tilde{v}^S + \varepsilon_2$ in which $\tilde{v}^S$ is the censored random wage derived by discarding the realized wages of movers
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Interpretation \& Policy}

\begin{frame}{Policy Angles \& Extensions}

  \end{itemize}

\begin{frame}{Value Function \& Choice Probabilities \textit{(Appendix)}}
  \begin{block}{Bellman equation}
\begin{align*}
V(x,\boldsymbol{\zeta}) &= \max_{j\in \mathcal{J}} \{\, v(x,j) + \zeta_j \,\},\\
v(x,j) &= u(x,j) + \beta \mathbb{E}_{x'}[\, \bar V(x') \mid x,j\,], \quad \bar V(x)=\mathbb{E}_{\zeta}[V(x,\zeta)].
\end{align*}
\end{block}
\begin{block}{Extreme value shocks $\Rightarrow$ logit choice probabilities}
\begin{equation*}
\rho(x,j)=\frac{\exp(v(x,j))}{\sum_{k\in\mathcal{J}} \exp(v(x,k))},\qquad
\exp(\bar V(x))=e^{\bar\gamma}\sum_{k}\exp(v(x,k)).
\end{equation*}
\end{block}
\end{frame}

\begin{frame}{Limited-History State Transitions \textit{(Appendix)}}
  \begin{block}{State and transitions (with $M=2$ recent locations)}
State: $x=(\ell_0,\ell_1,\ \omega_{\ell_0}^{(w)},\omega_{\ell_1}^{(w)},\ \omega_{\ell_0}^{(\xi)},\omega_{\ell_1}^{(\xi)},\ a).$\\[0.3em]
If stay: $x'=(\ell_0,\ell_1,\ \cdot,\cdot,\cdot,\cdot,\ a+1).$\\
If move to previous $\ell_1$: swap $(\ell_0,\ell_1)$ and their components, $a\leftarrow a+1.$\\
If move to new $j$: shift $(\ell_0\!\leftarrow\! j,\ \ell_1\!\leftarrow\! \ell_0)$ and draw new $(\omega_j^{(w)},\omega_j^{(\xi)})$.
\end{block}
\end{frame}

\begin{frame}{Likelihood \& Identification Sketch \textit{(Appendix)}}
  \begin{itemize}
    \item Joint likelihood of income \& choices; mixture over unobserved components.
    \item Nonparametric intuition: selection on bad current draws $\uparrow$ move probability.
  \end{itemize}
\end{frame}
